# Hash rules and the Lua/C++ binding surface, excluding art/audio and user configuration.
file(GLOB_RECURSE QS_RULE_FILES CONFIGURE_DEPENDS src/core/*.cpp src/core/*.h src/server/*.cpp src/server/*.h src/package/*.cpp src/package/*.h src/scenario/*.cpp src/scenario/*.h swig/*.i assets/game/lua/*.lua assets/game/etc/customScenes/*.txt patches/*.json)
list(SORT QS_RULE_FILES)
set(QS_RULE_INPUT "")
foreach(rule IN LISTS QS_RULE_FILES)
 file(SHA256 "${rule}" digest)
 file(RELATIVE_PATH name "${PROJECT_SOURCE_DIR}" "${rule}")
 string(APPEND QS_RULE_INPUT "${name}:${digest}\n")
endforeach()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${QS_RULE_FILES})
string(SHA256 QS_RULES_FINGERPRINT "${QS_RULE_INPUT}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/qs_build_identity.h" "#pragma once\n#define QS_RULES_FINGERPRINT \"${QS_RULES_FINGERPRINT}\"\n")
target_include_directories(qs_legacy_objects PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
