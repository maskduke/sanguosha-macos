cmake_minimum_required(VERSION 3.25)
project(QSanguoshaNirvana VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
option(QS_ENABLE_ASAN "Instrument native tests and libraries" OFF)
if(QS_ENABLE_ASAN)
  add_compile_options(-fsanitize=address -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address)
endif()
find_package(Qt6 6.11.1 EXACT REQUIRED COMPONENTS Core Test)
include(CTest)

# Preserve the matched Lua 5.1 sources; avoid the removed legacy dyld API.
file(GLOB LUA_SOURCES CONFIGURE_DEPENDS src/lua/*.c)
add_library(qs_lua STATIC ${LUA_SOURCES})
set_target_properties(qs_lua PROPERTIES AUTOMOC OFF C_STANDARD 99)
target_include_directories(qs_lua PUBLIC src/lua)
target_compile_definitions(qs_lua PRIVATE LUA_USE_POSIX LUA_USE_DLOPEN)

add_library(qs_protocol STATIC src/platform/network/line_framer.cpp)
target_include_directories(qs_protocol PUBLIC src)
target_link_libraries(qs_protocol PUBLIC Qt6::Core)

if(BUILD_TESTING)
  add_executable(test_line_framer tests/protocol/test_line_framer.cpp)
  target_link_libraries(test_line_framer PRIVATE qs_protocol Qt6::Test)
  add_test(NAME protocol.line_framer COMMAND test_line_framer)
  add_executable(test_lua_runtime tests/bindings/test_lua_runtime.cpp)
  target_link_libraries(test_lua_runtime PRIVATE qs_lua Qt6::Test)
  target_compile_definitions(test_lua_runtime PRIVATE QS_ASSET_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/assets/game")
  add_test(NAME bindings.lua_runtime COMMAND test_lua_runtime)
  find_path(VORBIS_INCLUDE_DIR vorbis/vorbisfile.h HINTS /opt/homebrew/opt/libvorbis/include REQUIRED)
  find_library(VORBISFILE_LIBRARY vorbisfile HINTS /opt/homebrew/opt/libvorbis/lib REQUIRED)
  find_path(OGG_INCLUDE_DIR ogg/ogg.h HINTS /opt/homebrew/opt/libogg/include REQUIRED)
  add_executable(test_original_audio tests/assets/test_original_audio.cpp)
  target_include_directories(test_original_audio PRIVATE ${VORBIS_INCLUDE_DIR} ${OGG_INCLUDE_DIR})
  target_link_libraries(test_original_audio PRIVATE Qt6::Test ${VORBISFILE_LIBRARY})
  target_compile_definitions(test_original_audio PRIVATE
    QS_ASSET_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/assets/game"
    QS_MANIFEST="${CMAKE_CURRENT_SOURCE_DIR}/docs/baseline/original-manifest.json")
  add_test(NAME assets.original_audio COMMAND test_original_audio)
  set_tests_properties(protocol.line_framer bindings.lua_runtime PROPERTIES TIMEOUT 30)
  set_tests_properties(assets.original_audio PROPERTIES TIMEOUT 180)
endif()

include(cmake/LegacyCompileProbe.cmake)
if(BUILD_TESTING)
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/qt_values_wrap.cxx"
    COMMAND ${SWIG_EXECUTABLE} -c++ -lua -I${PROJECT_SOURCE_DIR}/swig
      -o "${CMAKE_CURRENT_BINARY_DIR}/qt_values_wrap.cxx" "${PROJECT_SOURCE_DIR}/tests/bindings/qt_values.i"
    DEPENDS tests/bindings/qt_values.i swig/naturalvar.i swig/qvariant_core.i VERBATIM)
  add_executable(test_qt_values tests/bindings/test_qt_values.cpp "${CMAKE_CURRENT_BINARY_DIR}/qt_values_wrap.cxx")
  target_link_libraries(test_qt_values PRIVATE qs_lua Qt6::Test)
  add_test(NAME bindings.qt_values COMMAND test_qt_values)
  set_tests_properties(bindings.qt_values PROPERTIES TIMEOUT 30)
endif()

# The historical JSON layer remains separate from the new transport framing.
file(GLOB LEGACY_JSON_SOURCES CONFIGURE_DEPENDS src/jsoncpp/src/*.cpp)
add_library(qs_legacy_protocol STATIC src/core/jsonutils.cpp src/core/protocol.cpp ${LEGACY_JSON_SOURCES})
target_include_directories(qs_legacy_protocol PUBLIC src/core src/jsoncpp/include)
target_link_libraries(qs_legacy_protocol PUBLIC Qt6::Core)
if(BUILD_TESTING)
  add_executable(test_legacy_protocol tests/protocol/test_legacy_protocol.cpp)
  target_link_libraries(test_legacy_protocol PRIVATE qs_legacy_protocol Qt6::Test)
  add_test(NAME protocol.legacy_json COMMAND test_legacy_protocol)
  set_tests_properties(protocol.legacy_json PROPERTIES TIMEOUT 30)
endif()
